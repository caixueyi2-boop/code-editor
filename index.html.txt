    <script>
        const htmlCode = document.getElementById('htmlCode');
        const cssCode = document.getElementById('cssCode');
        const jsCode = document.getElementById('jsCode');
        const cppCode = document.getElementById('cppCode');
        const preview = document.getElementById('preview');
        const runBtn = document.getElementById('runBtn');

        function updatePreview() {
            const html = htmlCode.value;
            const css = `<style>${cssCode.value}</style>`;
            const js = `<script>${jsCode.value}<\/script>`;

            const output = `
                <!DOCTYPE html>
                <html>
                <head>
                    <meta charset="UTF-8">
                    ${css}
                </head>
                <body>
                    ${html}
                    ${js}
                </body>
                </html>
            `;

            const iframeDoc = preview.contentDocument || preview.contentWindow.document;
            iframeDoc.open();
            iframeDoc.write(output);
            iframeDoc.close();
        }

        runBtn.addEventListener('click', updatePreview);
        htmlCode.addEventListener('input', updatePreview);
        cssCode.addEventListener('input', updatePreview);
        jsCode.addEventListener('input', updatePreview);

        function handleFileUpload(inputElement, targetTextArea) {
            inputElement.addEventListener('change', (event) => {
                const file = event.target.files[0];
                if (!file) return;
                const reader = new FileReader();
                reader.onload = function(e) {
                    targetTextArea.value = e.target.result;
                    // 如果上傳的是網頁相關檔案，順便更新預覽
                    if (targetTextArea !== cppCode) {
                        updatePreview();
                    }
                };
                reader.readAsText(file);
            });
        }

        handleFileUpload(document.getElementById('uploadHtml'), htmlCode);
        handleFileUpload(document.getElementById('uploadCss'), cssCode);
        handleFileUpload(document.getElementById('uploadJs'), jsCode);
        handleFileUpload(document.getElementById('uploadCpp'), cppCode);

        updatePreview();
    </script>
